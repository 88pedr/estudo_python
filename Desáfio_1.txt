print("Quero ver se você consegue pular um muro com uma altura de 2 metros")
altura_base=input("Qual sua altura")
idade_base=input("Qual sua idade")
peso_base=input("Qual seu peso?")


def calcular():
	altura=float(altura_base)
	idade= float(idade_base)
	peso=int(peso_base)
	imc = peso/(altura**2)
	if (imc >= 18.5) and (altura >= 1.70) and (idade >= 14):
		print("Você consegue pular o muro")
	elif (idade < 14):
		print("Você é muito jovem para pular o muro")
	elif (altura < 1.70):
		print("você é pequenino")
	elif (imc <=18.5):
		print("Você não está saudavel o suficiente")
	elif (idade > 59):
		print("Está velho demais para isso")
	else:
		print("Você não consegue pular o muro")
calcular()
